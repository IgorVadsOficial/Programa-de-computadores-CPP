#include <iostream>
#include <fstream>
#include <string>

int main() {
    std::ifstream arquivo("meu_arquivo.txt"); // Abre o arquivo em modo leitura

    if (!arquivo.is_open()) {
        std::cerr << "Erro ao abrir o arquivo!\n";
        return 1;
    }

    std::string linha;
    while (std::getline(arquivo, linha)) {
        std::cout << linha << std::endl;
    }

    arquivo.close();
    return 0;
}



#include <iostream>
#include <fstream>

int main() {
    std::ofstream arquivo("meu_arquivo.txt"); // Abre o arquivo em modo escrita (sobrescreve)
    std::ofstream arquivo("meu_arquivo.txt",std::ios::app); // Sem sobrescrever o arquivo


    if (!arquivo.is_open()) {
        std::cerr << "Erro ao abrir o arquivo!\n";
        return 1;
    }

    arquivo << "Esta é a primeira linha do meu arquivo.\n";
    arquivo << "E esta é a segunda linha.\n";

    arquivo.close();
    return 0;
}
